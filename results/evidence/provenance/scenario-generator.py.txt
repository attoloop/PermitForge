def bake():
    RAW.mkdir(parents=True, exist_ok=True)
    def fetch(item):
        name, expected = item
        path = RAW / name
        for attempt in range(3):
            try:
                digest = hashlib.sha256()
                with urllib.request.urlopen(asset_url(name), timeout=180) as response, path.open('wb') as target:
                    while block := response.read(1024 * 1024):
                        target.write(block)
                        digest.update(block)
                if digest.hexdigest() != expected:
                    raise RuntimeError(f'NOAA snapshot changed: {name}; expected {expected}, got {digest.hexdigest()}')
                return
            except Exception:
                if attempt == 2:
                    raise
    with concurrent.futures.ThreadPoolExecutor(max_workers=4) as pool:
        list(pool.map(fetch, sorted(ASSETS.items())))
    write_json(RAW / 'sha256.json', ASSETS)


def run(argv, cwd, timeout, logname='console.txt'):
    with (cwd / logname).open('w') as log:
        try:
            subprocess.run(argv, cwd=cwd, stdout=log, stderr=subprocess.STDOUT,
                           timeout=timeout, check=True)
        except (subprocess.CalledProcessError, subprocess.TimeoutExpired) as exc:
            log.flush()
            raise RuntimeError(f'{Path(argv[0]).name} failed: {exc}; ' + (cwd / logname).read_text()[-5000:]) from exc


def compile_models(work):
    root = Path('/products') / PRODUCT
    manifest = json.loads((root / 'manifest.json').read_text())
    bindir = work / 'bin'
    bindir.mkdir()
    exes = {}
    for model, unit_type in [('aermet', 'aermet-source-tree'), ('aermod', 'aermod-source-tree'), ('bpip', 'bpip-prime-source-tree')]:
        units = [u for u in manifest['units'] if u['unit_type'] == unit_type]
        if len(units) != 1:
            raise RuntimeError(f'Expected one {unit_type} in runtime product')
        src = root / units[0]['content_path']
        if (src / 'content').is_dir():
            src = src / 'content'
        dest = bindir / model
        dest.mkdir()
        for path in sorted(src.rglob('*')):
            if path.is_file() and path.suffix.lower() in ['.f', '.f90', '.for', '.bat']:
                shutil.copyfile(path, dest / path.name)
        exes[model] = str(bindir / (model + '.exe'))
        if model == 'bpip':
            path = dest / 'Bpipprm.for'
            path.write_bytes(path.read_bytes().replace(b'\x1a', b''))
            args = ['-O2', '-static', '-std=legacy', '-fallow-argument-mismatch', 'Bpipprm.for']
        else:
            bat = next(dest.glob('gfortran*.bat')).read_text()
            order = re.findall(r'^gfortran -c %COMPILE_FLAGS%\s+(\S+)', bat, re.M)
            if not order:
                raise RuntimeError(f'No EPA compilation order for {model}')
            if model == 'aermet':
                args = ['-O2', '-static', '-fcheck=all', '-std=f2008', '-ffree-form', *order]
            else:
                args = ['-O2', '-static', '-fbounds-check', *order]
        run(['gfortran', *args, '-o', exes[model]], dest, 240)
    return exes


def clean_report(text):
    # Remove execution clock headers, preserving meteorological dates and model version.
    lines = []
    for line in text.replace('\r\n', '\n').splitlines():
        if re.match(r'\s*(DATE\s*:|TIME\s*:)', line):
            continue
        if re.match(r'\s*\*+ AERM(?:OD|ET)\b', line):
            line = re.sub(r'\s+\d{1,2}/\d{1,2}/\d{2,4}\s*$', '', line)
            line = re.sub(r'\s+\d{1,2}:\d{2}:\d{2}\s*$', '', line)
        if 'PROCESSING DATE/TIME:' in line:
            continue
        lines.append(line.rstrip())
    return '\n'.join(lines) + '\n'


def make_met(s, cache, exes):
    directory = cache / s['id']
    if directory.exists():
        return directory
    directory.mkdir(parents=True)
    with (directory / 'surface.dat').open('wb') as target:
        for year in range(2020, 2025):
            with gzip.open(RAW / f'{s["usaf"]}-{s["wban"]}-{year}.gz') as source:
                shutil.copyfileobj(source, target)
    with zipfile.ZipFile(RAW / (s['ua'] + '-data.txt.zip')) as archive, (directory / 'upper.dat').open('wb') as target:
        with archive.open(s['ua'] + '-data.txt') as source:
            keep = False
            for line in source:
                if line.startswith(b'#'):
                    keep = 2020 <= int(line[13:17]) <= 2024
                if keep:
                    target.write(line)
    albedo, bowen, roughness = LAND_USE[s['land_use']]
    inp = f'''JOB
 REPORT report.txt
 MESSAGES messages.txt
UPPERAIR
 DATA upper.dat IGRA
 LOCATION {s['ua_id']} {s['ua_lat']}N {abs(s['ua_lon'])}W {s['tz']} {s['ua_elev']}
 XDATES 2020/1/1 TO 2024/12/31
 MODIFY
SURFACE
 DATA surface.dat ISHD
 LOCATION {s['wban']} {s['lat']}N {abs(s['lon'])}W {s['tz']} {s['elev']}
 XDATES 2020/1/1 TO 2024/12/31
METPREP
 XDATES 2020/1/1 TO 2024/12/31
 OUTPUT {s['id']}.sfc
 PROFILE {s['id']}.pfl
 METHOD WIND_DIR RANDOM
 METHOD REFLEVEL SUBNWS
 NWS_HGT WIND {s['wind_height_m']}
 FREQ_SECT ANNUAL 1
 SECTOR 1 0 360
 SITE_CHAR 1 1 {albedo} {bowen} {roughness}
'''
    (directory / 'aermet.inp').write_text(inp)
    run([exes['aermet'], 'aermet.inp'], directory, 180)
    if 'AERMET FINISHED SUCCESSFULLY' not in (directory / 'console.txt').read_text():
        raise RuntimeError(f'AERMET failed for {s["id"]}: ' + (directory / 'messages.txt').read_text()[-5000:])
    rows = (directory / (s['id'] + '.sfc')).read_text().splitlines()[1:]
    if len(rows) != 43848 or not rows[0].startswith('2020') or not rows[-1].startswith('2024'):
        raise RuntimeError(f'Incomplete five-year AERMET output for {s["id"]}')
    for name in ['surface.dat', 'upper.dat', 'console.txt', 'messages.txt', 'report.txt']:
        (directory / name).unlink()
    return directory


def receptors():
    points = set()
    for n in range(-250, 251, 50):
        points.update([(n, -250), (n, 250), (-250, n), (250, n)])
    for extent, step in [(1000, 250), (5000, 1000)]:
        for x in range(-extent, extent + 1, step):
            for y in range(-extent, extent + 1, step):
                if max(abs(x), abs(y)) >= 250 and x*x + y*y <= 5000*5000:
                    points.add((x, y))
    assert len(points) <= 1500
    return sorted(points)


def draw_facility(rng, archetype):
    buildings = []
    for j, (x, y) in enumerate([(-90, -90), (90, -90), (-90, 90), (90, 90)][:rng.randint(1, 4)]):
        w, d, h = [round(rng.uniform(a, b), 2) for a, b in [(25, 65), (25, 65), (12, 40)]]
        buildings.append({'id': f'B{j+1}', 'base_elevation_m': 0, 'height_m': h,
                          'corners_m': [[x-w/2, y-d/2], [x+w/2, y-d/2], [x+w/2, y+d/2], [x-w/2, y+d/2]]})
    sources = []
    for j in range(rng.randint(2, 6)):
        b = buildings[j % len(buildings)]
        x = b['corners_m'][1][0] + rng.uniform(3, 12)
        y = (b['corners_m'][1][1] + b['corners_m'][2][1]) / 2 + rng.uniform(-10, 10)
        kind = archetype
        if archetype == 'small_refinery_mix':
            kind = ['gas_boilers', 'process_heaters', 'small_refinery_mix'][j % 3]
        ranges = ARCHETYPES[kind]
        values = [round(rng.uniform(a, b), 6) for a, b in ranges]
        sources.append({'id': f'S{j+1}', 'type': kind, 'location': {'x_m': round(x, 2), 'y_m': round(y, 2), 'base_elevation_m': 0},
                        'uncontrolled_rates_g_s': dict(zip(['NO2', 'SO2', 'PM25'], values[:3])),
                        'no2_nox_in_stack_ratio': round(rng.uniform(0.08, 0.25), 4),
                        'stack_bounds': {'height_min_m': round(rng.uniform(8, min(25, b['height_m'])), 2)},
                        'fixed_stack': {'temperature_K': values[3], 'velocity_m_s': values[4], 'diameter_m': values[5]}})
    return buildings, sources


def bpip_input(buildings, sources, heights):
    lines = ["'Foundry synthetic facility'", "'P'", "'METERS' 1.0", "'UTMN' 0.0", str(len(buildings))]
    for b in buildings:
        lines += [f"'{b['id']}' 1 0", f"4 {b['height_m']}"]
        lines += [f'{x:.4f} {y:.4f}' for x, y in b['corners_m']]
    lines.append(str(len(sources)))
    for u in sources:
        p = u['location']
        lines.append(f"'{u['id']}' 0 {heights[u['id']]} {p['x_m']} {p['y_m']}")
    return '\n'.join(lines) + '\n'


def run_bpip(directory, buildings, sources, heights, exes):
    directory.mkdir(parents=True, exist_ok=True)
    text = bpip_input(buildings, sources, heights)
    (directory / 'bpip.inp').write_text(text)
    (directory / 'fort.10').write_text(text)
    run([exes['bpip']], directory, 60, 'bpip.console')
    report = (directory / 'fort.12').read_text()
    cards = [line.strip() for line in report.splitlines() if line.strip().startswith('SO ')]
    if len(cards) != len(sources) * 30:
        raise RuntimeError('BPIP did not emit all 36-direction PRIME parameters')
    gep = {}
    table = report.split('PRELIMINARY* GEP STACK HEIGHT RESULTS TABLE', 1)[1].split('* Results', 1)[0]
    for line in table.splitlines():
        fields = line.split()
        if len(fields) == 5 and fields[0] in heights:
            gep[fields[0]] = max(65.0, float(fields[-1]))
    if set(gep) != set(heights):
        raise RuntimeError('BPIP GEP results missing a source')
    (directory / 'bpip.out').write_text(clean_report(report))
    (directory / 'bpip.sum').write_text(clean_report((directory / 'fort.14').read_text()))
    for name in ['fort.10', 'fort.12', 'fort.14', 'bpip.console']:
        (directory / name).unlink()
    return '\n'.join(cards), gep


def concentration_rows(path):
    with path.open() as source:
        for line in source:
            if line.strip() and not line.startswith('*'):
                fields = line.split()
                x, y, value = map(float, fields[:3])
                if not math.isfinite(value) or value < 0:
                    raise RuntimeError(f'Invalid concentration in {path.name}')
                yield fields, x, y, value


def max_concentration(path, expected):
    rows = list(concentration_rows(path))
    if len(rows) != expected:
        raise RuntimeError(f'Unexpected receptor/year count in {path.name}: {len(rows)}')
    return max(row[3] for row in rows)


def so2_three_hour(path, nrec):
    # POSTFILE 3 PLOT records every nonoverlapping 3-hour block. Track each
    # receptor's two highest block means within each calendar year.
    highs = {}
    count = 0
    for fields, x, y, value in concentration_rows(path):
        year = 2000 + int(fields[8][:2])
        key = (year, x, y)
        a, b = highs.get(key, (-1.0, -1.0))
        if value >= a:
            a, b = value, a
        elif value > b:
            b = value
        highs[key] = (a, b)
        count += 1
    if set(k[0] for k in highs) != set(range(2020, 2025)) or len(highs) != 5 * nrec or min(v[1] for v in highs.values()) < 0:
        raise RuntimeError('Incomplete annual SO2 3-hour statistics')
    summaries = [{'year': year, 'x_m': x, 'y_m': y, 'second_highest_ug_m3': vals[1]} for (year, x, y), vals in sorted(highs.items())]
    write_json(path.with_name('annual_second_highest.json'), summaries)
    path.unlink()
    return max(v[1] for v in highs.values())


def run_models(root, content, s, sources, heights, rates, cards, tier, ozone, exes):
    results = {}
    nrec = len(receptors())
    configs = [('NO2', root, '1 ANNUAL', '1', '8TH'), ('PM25', root / 'pm25', '24 ANNUAL', '24', '8TH'),
               ('SO2', root / 'so2', '1 ANNUAL', '1', '4TH'), ('SO2', root / 'so2_3h', '3', '3', '2ND')]
    def model(config):
        pollutant, directory, averages, period, rank = config
        directory.mkdir(exist_ok=True)
        metpath = os.path.relpath(content / 'met' / s['id'], directory)
        recpath = os.path.relpath(content / 'receptors.inp', directory)
        options = 'DFAULT CONC' + (' ' + tier if pollutant == 'NO2' else '')
        lines = ['CO STARTING', 'CO TITLEONE Foundry synthetic permit facility', f'CO MODELOPT {options}',
                 f'CO AVERTIME {averages}', f'CO POLLUTID {pollutant}', 'CO RUNORNOT RUN']
        if pollutant == 'NO2' and tier == 'PVMRM':
            lines.append(f'CO OZONEVAL {ozone} PPB')
        lines += ['CO FINISHED', 'SO STARTING']
        for u in sources:
            name = u['id']; p = u['location']; st = u['fixed_stack']
            lines += [f"SO LOCATION {name} POINT {p['x_m']} {p['y_m']} 0",
                      f"SO SRCPARAM {name} {rates[name][pollutant]:.9f} {heights[name]} {st['temperature_K']} {st['velocity_m_s']} {st['diameter_m']}"]
            if pollutant == 'NO2' and tier == 'PVMRM':
                lines.append(f"SO NO2RATIO {name} {u['no2_nox_in_stack_ratio']}")
        lines += [cards, 'SO SRCGROUP ALL', 'SO FINISHED', 'RE STARTING', f'RE INCLUDED {recpath}', 'RE FINISHED',
                  'ME STARTING', f'ME SURFFILE {metpath}.sfc', f'ME PROFFILE {metpath}.pfl',
                  f"ME SURFDATA {s['wban']} 2020", f"ME UAIRDATA {s['ua_id']} 2020", 'ME PROFBASE 0', 'ME FINISHED',
                  'OU STARTING', f'OU RECTABLE {period} {rank}', f'OU PLOTFILE {period} ALL {rank} short.plt']
        if 'ANNUAL' in averages:
            if pollutant == 'NO2':
                lines.append('OU POSTFILE ANNUAL ALL PLOT annual.plt')
            else:
                lines.append('OU PLOTFILE ANNUAL ALL annual.plt')
        if directory.name == 'so2_3h':
            lines.append('OU POSTFILE 3 ALL PLOT blocks.plt')
        lines.append('OU FINISHED')
        (directory / 'aermod.inp').write_text('\n'.join(lines) + '\n')
        run([exes['aermod']], directory, 1800)
        output = (directory / 'aermod.out').read_text()
        if 'AERMOD Finishes Successfully' not in output or not re.search(r'43848 Hours Were Processed', output):
            raise RuntimeError(f'AERMOD {pollutant} {averages} failed: ' + output[-6000:])
        if directory.name == 'so2_3h':
            results['SO2_3h'] = so2_three_hour(directory / 'blocks.plt', nrec)
        else:
            results[pollutant + ('_24h' if pollutant == 'PM25' else '_1h')] = max_concentration(directory / 'short.plt', nrec)
            annual = max_concentration(directory / 'annual.plt', nrec * (5 if pollutant == 'NO2' else 1))
            if pollutant == 'PM25':
                results['PM25_annual_primary'] = annual
                results['PM25_annual_secondary'] = annual
            else:
                results[pollutant + '_annual'] = annual
        for path in sorted(directory.iterdir()):
            if path.is_file() and path.suffix in ['.out', '.plt']:
                path.write_text(clean_report(path.read_text()))
        for path in list(directory.iterdir()):
            if path.is_file() and path.name not in ['aermod.inp', 'aermod.out', 'short.plt', 'annual.plt', 'annual_second_highest.json', 'bpip.inp', 'bpip.out', 'bpip.sum']:
                path.unlink()
    with concurrent.futures.ThreadPoolExecutor(max_workers=4) as pool:
        list(pool.map(model, configs))
    if set(results) != set(STANDARDS):
        raise RuntimeError('Missing admission statistics')
    return results


def generate_one(seed, index, work, exes):
    digest = hashlib.sha256(seed.encode('utf-8') + str(index).encode('ascii')).hexdigest()
    rng = random.Random(int(digest, 16))
    s = rng.choice(STATIONS)
    archetype = rng.choice(sorted(ARCHETYPES))
    # Prefix-stable schedule guarantees >=70% Tier 3 for every positive COUNT.
    tier = 'PVMRM' if index % 10 < 7 else 'ARM2'
    met = make_met(s, work / 'met', exes)
    for attempt in range(32):
        candidate = work / 'candidate'
        if candidate.exists():
            shutil.rmtree(candidate)
        content = candidate / 'content'
        content.mkdir(parents=True)
        shutil.copytree(met, content / 'met')
        buildings, sources = draw_facility(rng, archetype)
        write_json(content / 'buildings.json', {'coordinate_system': 'local Cartesian meters; flat terrain', 'buildings': buildings})
        (content / 'receptors.inp').write_text(''.join(f'RE DISCCART {x} {y} 0 0\n' for x, y in receptors()))
        minimum = {u['id']: u['stack_bounds']['height_min_m'] for u in sources}
        uncontrol = {u['id']: u['uncontrolled_rates_g_s'] for u in sources}
        costs = {'currency': 'USD', 'basis': 'synthetic annualized engineering costs; no real permit data',
                 'stack_usd_per_m': round(rng.uniform(1500, 6500), 2), 'controls': []}
        control_map = {}
        efficiencies = {'NO2': 0.95, 'SO2': 0.98, 'PM25': 0.995}
        for u in sources:
            maximum_cost = rng.uniform(120000, 500000)
            for label, eff, cost_scale in [('none', 0, 0), ('moderate', 0.75, 0.55), ('maximum', 1, 1)]:
                cid = u['id'] + '_' + label
                costs['controls'].append({'id': cid, 'applies_to': [u['id']],
                    'efficiency_by_pollutant': {p: round(v * eff, 6) for p, v in efficiencies.items()},
                    'annualized_usd': round(maximum_cost * cost_scale, 2)})
                if label == 'maximum':
                    control_map[u['id']] = cid

            u['stack_bounds']['gep_height_m'] = gep[u['id']]
        draft_cards, draft_gep = run_bpip(draft, buildings, sources, gep, exes)
        if draft_gep != gep:
            raise RuntimeError('GEP changed with stack height')
        controlled = {u['id']: {p: round(rate * (1 - efficiencies[p]), 9) for p, rate in u['uncontrolled_rates_g_s'].items()} for u in sources}
        print(f'unit {index}, candidate {attempt}: {s["id"]} {archetype} {tier}', file=sys.stderr, flush=True)
        baseline = run_models(base, content, s, sources, minimum, uncontrol, base_cards, tier, ozone, exes)
        base_values = {k: baseline[k] + background[k] for k in STANDARDS}
        base_ratios = {k: base_values[k] / STANDARDS[k]['limit'] for k in STANDARDS}
        if max(base_ratios.values()) <= 1:
            continue
        designed = run_models(draft, content, s, sources, gep, controlled, draft_cards, tier, ozone, exes)
        design_values = {k: designed[k] + background[k] for k in STANDARDS}
        design_ratios = {k: design_values[k] / STANDARDS[k]['limit'] for k in STANDARDS}
        if max(design_ratios.values()) > 1:
            continue

        write_json(draft / 'admission.json', {'modeled_ug_m3': designed, 'total_with_background_ug_m3': design_values, 'ratios': design_ratios})
        controls_cost = sum(c['annualized_usd'] for c in costs['controls'] if c['id'] in control_map.values())
        draft_cost = round(controls_cost + sum(gep[k] - minimum[k] for k in gep) * costs['stack_usd_per_m'], 2)
        design = {'units': [{'id': u['id'], 'control_id': control_map[u['id']], 'stack_height_m': gep[u['id']], 'rates_g_s': controlled[u['id']]} for u in sources],
                  'annualized_cost_usd': draft_cost, 'design_values_ug_m3': design_values}
        write_yaml(draft / 'design.yaml', design)
        write_json(content / 'cost_table.json', costs)
        write_json(content / 'background.json', {'units': 'ug/m3', 'design_values': background, 'basis': 'seeded synthetic design-value backgrounds',

                    'terrain': 'flat', 'dispersion_setting': 'rural', 'synthetic': True}, 'units': sources,

                    'naaqs_thresholds': STANDARDS, 'naaqs_source': 'https://www.epa.gov/criteria-air-pollutants/naaqs-table',
                    'naaqs_scope': 'current standards for modeled pollutants plus retained 3-hour SO2; no area-specific revoked standards',
                    'locked_keywords': {'MODELOPT': ['DFAULT', 'CONC'], 'NO2_tier': tier, 'ozone_ppb': ozone, 'meteorology_years': [2020, 2021, 2022, 2023, 2024],
                                        'receptors': 'receptors.inp', 'buildings': 'buildings.json', 'in_stack_ratios': 'units[].no2_nox_in_stack_ratio'},
                    'model_runs': ['aermod.inp', 'pm25/aermod.inp', 'so2/aermod.inp', 'so2_3h/aermod.inp'],
                    'run_directory': 'draft', 'surface_characteristics': {'assumption': 'fixed land-use class, not AERSURFACE or a site survey',

                'no2_tier': tier, 'no2_limited': limiting.startswith('NO2'),
                'downwash_sensitive': any(minimum[k] < gep[k] for k in minimum),
                'downwash_sensitive_definition': 'geometric screening flag: minimum stack height below BPIP GEP; not a no-building counterfactual',
                'background_fraction': background[limiting] / STANDARDS[limiting]['limit'],
                'background_fractions': {k: background[k] / v['limit'] for k, v in STANDARDS.items()},
                'runtime_product': PRODUCT, 'versions': {'AERMOD': '26135', 'AERMET': '26135', 'BPIPPRM': '04274'},
                'noaa_snapshots': [{'name': name, 'url': asset_url(name), 'sha256': ASSETS[name]} for name in relevant_assets],
                'determinism': 'Random seed is sha256(UTF-8 SEED concatenated with ASCII index); AERMET uses its fixed EPA wind-direction table',

                'SO2_3h_statistic': 'second-highest 3h block per receptor per calendar year from POSTFILE, then maximum',
                'n_receptors': len(receptors())}

    raise RuntimeError(f'No admissible candidate for index {index} after 32 deterministic draws')


def main():
    if sys.argv[1:] == ['--bake']:
        bake()
        return
    if sys.argv[1:]:
        raise RuntimeError('Unexpected arguments; runtime takes SEED and COUNT environment variables')
    count = int(os.environ.get('COUNT') or '360')
    if count < 0:
        raise RuntimeError('COUNT must be nonnegative')
    # RESUME=1 (datafoundry.resume): /output holds only finished units, each
    # published by one rename; skip them and make the rest.
    resume = os.environ.get('RESUME') == '1'
    present = list(OUT.iterdir())
    if present and not resume:
        raise RuntimeError('/output must initially be empty')
    stray = [p.name for p in present if not (p.is_dir() and re.fullmatch(r'scenario-\d{6}', p.name))]
    if stray:
        raise RuntimeError(f'with RESUME=1 /output may hold only finished scenario-NNNNNN directories: {stray[:3]}')
    if count == 0:
        return
    work = OUT / '.work'
    work.mkdir()
    (work / 'tmp').mkdir()
    os.environ['TMPDIR'] = str(work / 'tmp')
    os.environ['LC_ALL'] = 'C'
    os.environ['TZ'] = 'UTC'
    try:
        exes = compile_models(work)
        for index in range(count):
            if (OUT / f'scenario-{index:06d}').is_dir():
                continue
            generate_one(os.environ.get('SEED', '0'), index, work, exes)
    finally:
        shutil.rmtree(work)


if __name__ == '__main__':
    try:
        main()
    except Exception as exc:
        print(f'generator failed: {exc}', file=sys.stderr)
        sys.exit(1)
